MongoDB Shell Solutions
Database: assignment_7

Q1 - Create explicit books collection with validation
use assignment_7

db.createCollection("books", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["title"],
            properties: {
                title: {
                    bsonType: "string",
                    minLength: 1
                }
            }
        }
    },
    validationLevel: "strict",
    validationAction: "error"
})


Q2 - Create implicit authors collection by inserting data
db.authors.insertOne({
    name: "ahmed said2"
})


Q3 - Create capped logs collection (1 MB)
db.createCollection("logs", {
    capped: true,
    size: 1024 * 1024
})


Q4 - Create index on books.title
db.books.createIndex({
    title: 1
})


Q5 - Insert one book
db.books.insertOne({
    title: "Future",
    author: "Aldous Huxley",
    year: 2020,
    genres: ["Science Fiction"]
})


Q6 - Insert at least 3 books
db.books.insertMany([
    {
        title: "Brave New World",
        author: "Aldous Huxley",
        year: 1932,
        genres: ["Science Fiction", "Dystopian"]
    },
    {
        title: "The Martian",
        author: "Andy Weir",
        year: 2011,
        genres: ["Science Fiction"]
    },
    {
        title: "The Shining",
        author: "Stephen King",
        year: 1977,
        genres: ["Horror"]
    }
])


Q7 - Insert a new log
db.logs.insertOne({
    bookTitle: "Future",
    action: "Book created",
    timestamp: new Date()
})


Q8 - Update Future year to 2022
db.books.updateOne(
    { title: "Future" },
    { $set: { year: 2022 } }
)


Q9 - Find book by title
db.books.findOne({
    title: "Brave New World"
})


Q10 - Find books published between 1990 and 2010
db.books.find({
    year: {
        $gte: 1990,
        $lte: 2010
    }
})


Q11 - Find books whose genres include Science Fiction
db.books.find({
    genres: "Science Fiction"
})


Q12 - Skip first 2, limit next 3, sort year descending
db.books.find({})
    .sort({ year: -1 })
    .skip(2)
    .limit(3)


Q13 - Find books where year is stored as an integer
db.books.find({
    year: {
        $type: "int"
    }
})


Q14 - Find books excluding Horror and Science Fiction
db.books.find({
    genres: {
        $nin: ["Horror", "Science Fiction"]
    }
})


Q15 - Delete books published before 2000
db.books.deleteMany({
    year: {
        $lt: 2000
    }
})


Q16 - Aggregation: books after 2000, sorted by year descending
db.books.aggregate([
    {
        $match: {
            year: {
                $gt: 2000
            }
        }
    },
    {
        $sort: {
            year: -1
        }
    }
])


Q17 - Aggregation: books after 2000, show only title, author, year
db.books.aggregate([
    {
        $match: {
            year: {
                $gt: 2000
            }
        }
    },
    {
        $project: {
            _id: 0,
            title: 1,
            author: 1,
            year: 1
        }
    }
])


Q18 - Aggregation: unwind genres
db.books.aggregate([
    {
        $unwind: "$genres"
    }
])


Q19 - Aggregation: join books with logs
db.books.aggregate([
    {
        $lookup: {
            from: "logs",
            localField: "title",
            foreignField: "bookTitle",
            as: "logs"
        }
    }
])
